CREATE OR REPLACE FUNCTION GET_REFUNDED_AMOUNT(p VARCHAR, amount NUMBER)
RETURNS NUMBER
AS
$$
  CASE
    WHEN UPPER(p) = 'FULL' THEN amount
    ELSE 0
  END
$$;
